# 一、基础输入

## 1.1 LLM 层规格（`llm_models.py`）

```python
@dataclass
class LLMSpec:
    name: str
    layer_mem: np.ndarray      # 输入：c_{k,i} 各层内存 (MB)
    bytes_per_param: float = 2.0   # 输入：精度 b (fp16=2)，用于由内存反推参数量

    @property
    def flops_per_token(self):
        # ★ ψ_k = 2·N_params：每 token 计算量，【由内存反推】无需额外参数
        #   fp16 时恰好 ψ_k = C_{k,N}/1000；7B 得 14.73 GFLOP/token (参数量 7.36G)
        return 2.0 * (self.total_mem * 1e6 / self.bytes_per_param / 1e9)

    def __post_init__(self):
        # ★ 状态量：C_{k,s} = Σ_{i≤s} c_{k,i}（前 s 层累计内存，随 s 递增）
        self.C = np.concatenate([[0.0], np.cumsum(self.layer_mem)])
        #   C[0]=0（全卸载），C[N]=模型全常驻内存

    def reload_mem(self, s):
        # ★ 从状态 s 恢复到全常驻需重新载入的数据量 = C_{k,N} − C_{k,s}
        return self.total_mem - self.mem_at(s)
```

层顺序 = `[embedding, transformer × n, fc]`，与推理时的使用顺序一致 —— 这是"前缀常驻"假设的物理依据。

## 1.2 无线信道（`channel.py`）

```python
def uplink_rate_mbps(self, fading_pow=None):
    # ★ 香农速率 R_m = B log2(1 + P|h_m|²/(N0·B))
    b_hz = self.bandwidth_mhz * 1e6
    return b_hz * np.log2(1.0 + self.snr(fading_pow)) / 1e6

def path_gain(self, fading_pow=None):
    # ★ 信道增益 |h_m|² = g0 · ℓ_m^(−α) · |fading|²（距离越远增益越小）
    g0 = 10.0 ** (self.ref_gain_db / 10.0)
    large = g0 * self.distances_m ** (-self.path_loss_exp)
    return large if fading_pow is None else large * np.asarray(fading_pow)

def transmission_delay_ms(self, server, input_mbit, output_mbit, fading_pow=None):
    # ★ 传输时延 T^tx = S_in/R_ul + S_out/R_dl + 2·δ_m
    #   （即会议纪要里的"传输延迟/搬运成本" —— 二者是同一项，不是两项）
    t_ul = 1000.0 * input_mbit / self.uplink_rate_mbps(fading_pow)[server]
    t_dl = 1000.0 * output_mbit / self.downlink_rate_mbps(fading_pow)[server]
    return float(t_ul + t_dl + 2.0 * self.backhaul_delay_ms[server])
```

# 二、单服务器内核（`server_model.py`）

一台服务器 = 一块 GPU，托管 **K 个完整模型**，它们**共享同一块显存**。

## 2.1 输入 + 决策变量

```python
@dataclass
class ServerModel:
    models: list[LLMSpec]              # 系统中全部 K 个模型
    mem_cap_mb: float                  # 输入：显存容量 C̄（各服务器【相同】）
    load_bw_mb_per_ms: float           # 输入：存储→GPU 带宽 β_m（决定 d = c/β）
    compute_gflop_per_ms: float        # 输入：算力 f_m（各服务器【不同】，云最强）
    mem_bw_mb_per_ms: float            # 输入：显存带宽 ω_m（限制 decode 阶段）
    deadline_ms: np.ndarray            # 输入：T^max_k —— ★【唯一】的 QoS 参数
    hosts: np.ndarray | None = None    # ★ 放置变量 y_{m,k}：是否托管模型 k
    # ★ 决策变量 s_{m,k}(t)（保留轨迹）不作字段 —— 由 agent 动作给出（见第四节）
    # ★ d_qos_ms 不再是输入！由 deadline 推导（见 2.3），消除双重 QoS 的冗余
```

## 2.2 状态：保留状态的内存与加载延迟

```python
def loading_delay(self, k, s):
    # ★ 状态量：D_{m,k,s} = (C_{k,N} − C_{k,s}) / β_m
    #   模型 k 从保留状态 s 恢复到全常驻的加载时延（随 s 递减）
    #   ⚠ 用 ">s"（不含第 s 层）—— 状态 s 已保留第 s 层，只需重载其后的层。
    #     论文正文写 "≥i" 是笔误，只有 ">i" 能复现 TABLE I 的成本。
    return self.models[k].reload_mem(int(s)) / self.load_bw_mb_per_ms

def prefill_delay(self, k, in_tokens):
    # ★ prefill：一次并行处理全部输入 token => 受【算力】限制
    return self.models[k].flops_per_token * in_tokens / self.compute_gflop_per_ms

def decode_delay(self, k, out_tokens):
    # ★ decode：逐 token 生成，每步都要把全部权重从显存读一遍
    #   => 除算力外还受【显存带宽】限制，取二者较大值。
    #   实测三台服务器都取到第二项（带宽受限），与 LLM 服务的已知结论一致。
    spec = self.models[k]
    return out_tokens * max(spec.flops_per_token / self.compute_gflop_per_ms,
                            spec.total_mem / self.mem_bw_mb_per_ms)
```

## 2.3 约束（QoS）：最少常驻层数

```python
def _derive_d_qos(self):
    # ★ QoS 统一：加载预算 = deadline 扣掉传输与计算后的剩余
    #   D^QoS_{m,k} = T^max_k − T^tx_{k,m} − T^cp_{k,m}
    #   这样"满足 i_check" ⇒ "无排队时端到端必满足 deadline"，二者由构造自洽。
    #   涌现结论：算力强的服务器预算更大 ⇒ i_check 更小（实测 [19,14,3]，云最小）
    return self.deadline_ms - self.tx_overhead_ms - cp_per_type

def _compute_i_check(self):
    # ★ 约束（论文式 1a 的 per-(server,model) 版）：
    #   i_check_{m,k} = min{ s | D_{m,k,s} ≤ D^QoS_{m,k} }
    out = np.zeros(self.K, dtype=int)
    for k in range(self.K):
        if not self.hosts[k]:
            continue
        for s in range(self.models[k].N + 1):
            if self.loading_delay(k, s) <= self.d_qos_ms[k] + 1e-9:
                out[k] = s
                break
    return out
```

## 2.4 约束（共享显存）—— 新场景最主要的新增耦合 ⭐

```python
def total_mem(self, states):
    # ★ 当前所有托管模型的总显存占用 Σ_k C_{k,s_k}
    return float(sum(self.model_mem(k, states[k]) for k in range(self.K)))

def check_mem_cap(self, states):
    # ★ 约束 C3：Σ_k C_{k,s_{m,k}} ≤ C̄（K 个模型共享一块显存）
    return self.total_mem(states) <= self.mem_cap_mb + 1e-6

def is_feasible(self):
    # ★ 可行性条件(a)：QoS 要求的最小常驻集合能否装进显存
    #   —— 论文里不存在这个条件（它只有 1 个模型）
    return self.min_total_mem() <= self.mem_cap_mb + 1e-6

def can_serve(self, k):
    # ★ 可行性条件(b)：服务模型 k 需把它恢复到【全常驻】，其余最多压到各自 i_check
    #   C_{k,N} + Σ_{k'≠k} C_{k',i_check} ≤ C̄
    if not self.hosts[k]:
        return False
    need = self.models[k].total_mem
    others = self.min_total_mem() - self.model_mem(k, int(self.i_check[k]))
    return need + others <= self.mem_cap_mb + 1e-6
```

## 2.5 显存驱逐（服务某模型时给它腾位置）

```python
def make_room_for(self, states, k):
    # ★ 服务模型 k 前必须把它恢复全常驻；若超显存，驱逐【其他】模型的层
    #   驱逐规则：当前占用最大者优先降一层，且不低于各自 i_check（确定性、可复现）
    s = np.asarray(states, dtype=int).copy()
    s[k] = self.models[k].N
    while self.total_mem(s) > self.mem_cap_mb + 1e-6:
        cand = [kk for kk in range(self.K)
                if kk != k and self.hosts[kk] and s[kk] > self.i_check[kk]]
        if not cand:
            return None                      # 腾不出来（对应 can_serve(k)==False）
        k_evict = max(cand, key=lambda kk: self.model_mem(kk, s[kk]))
        s[k_evict] -= 1
    return s
```

## 2.6 数学钩子：ski-rental 临界点

```python
def _compute_B(self):
    # ★ B_{m,k} = η·D_{m,k,i_check} / (C_{k,N} − C_{k,i_check})
    #   代入 D=(ΔC)/β 得 B = η/β —— 与模型无关！
    #   这【自动复现了论文 C 节】"d=kc ⇒ 所有成本直线交于同一点 B=ηk"的结论。
    #   仅作数学钩子暴露（将来约束 DRL 动作空间），不用于在线决策。
    denom = spec.total_mem - spec.mem_at(int(self.i_check[k]))
    out[k] = self.eta * self.loading_delay(k, int(self.i_check[k])) / denom
```

```python
# system_multi_server.py —— ★ 权重的唯一来源
# η 不是自由参数！归一化目标下由权重与归一化常数共同决定：
#   η_k = (ω1/ω2) · τ·M·C̄ / T^max_k        (推导见第三章式 34)
# η_k ∝ 1/T^max_k 的含义：deadline 越紧 ⇒ 时延越贵 ⇒ 越该常驻。
# (审查 P0-3：曾出现 eta=30 与 ω1/ω2 两套权重，相差 333 倍)
def eta_for_model(self, k):
    return (self.w_lat / self.w_mem) * self.slot_ms * self.mem_ref / self.deadline_of(k)

for sv in self.servers:
    sv.set_eta(np.array([self.eta_for_model(k) for k in range(self.K)]))
```

# 三、全系统模型（`system_multi_server.py`）

## 3.1 四项时延（含排队）

```python
def task_latency(self, task, server, retention_state, wait_ms=0.0, fading_pow=None):
    # ★ T_j = T^wait + T^tx + T^ld + T^cp（四项，v4 新增排队等待）
    sv = self.servers[server]
    tx = self.channel.transmission_delay_ms(       # 传输（距离越远越大）
        server, task.input_mbit, task.output_mbit, fading_pow)
    ld = sv.loading_delay(task.model_id, retention_state)   # 加载（看模型是否常驻）
    cp = sv.compute_delay(task.model_id, task.in_tokens, task.out_tokens)  # prefill+decode
    return LatencyBreakdown(waiting=wait_ms, transmission=tx, loading=ld, computation=cp)
```

> 实测量级（7B, 512→128 token）：传输 2\~80 ms、计算 1274\~3354 ms、加载 0 或 2135\~4135 ms。
> 三项同量级且**方向相反**（近端传输小但计算慢、云反之）⇒ 重任务云最优、轻任务近端最优。


## 3.2 内存成本（已归一化）

```python
def normalized_latency(self, task, total_ms):
    # ★ Ψ̃ 的单项：按【各任务自己的 deadline】归一 ⇒ 值为 1 即恰好用满 QoS 预算
    return total_ms / task.deadline_ms

def normalized_slot_memory(self, states):
    # ★ Φ̃ 的单项：(Σ_m u_m)/(M·C̄) = 该时隙的跨服务器平均显存利用率 ∈ [0,1]
    #   mem_ref = Σ_m C̄ 在 __post_init__ 里算好，故每时隙贡献一个无量纲数
    return float(self.server_memory_usage(states).sum() / self.mem_ref)
```

> ⚠ v4 起**不再**用 `MB·ms` 的原始内存成本。三项原本量纲不同（ms / MB·ms / 无量纲），
> 直接相加没有物理意义；归一化后 ω 才是**可解释的纯权衡系数**（对抗性审查 ⑥）。

## 3.3 负载均衡（Jain 公平指数）

```python
def jain_index(loads):
    # ★ F = (Σ L_m)² / (M Σ L_m²) ∈ [1/M, 1]
    #   全相等 → 1（完美均衡）；全集中一台 → 1/M。沿用毕设做法。
    s = float(x.sum())
    if s <= 0:
        return 1.0                      # 约定：全零为平凡均衡 (对应式18的第二分支)
    return float(s ** 2 / (x.size * float((x ** 2).sum())))
```

```python
# env_multi_server.py · step()  —— ★ 负载 L_m(t) 的累加
# 必须计入【加载 + 计算 + 预取】：三者都占用 GPU (7B 上加载 2135 与计算 3351 ms 同量级)，
# 若只算计算时间，"频繁重载"的服务器会被误判为空闲，Jain 反而把更多任务推给它。
# (对抗性审查 P1-6；回归测试 loaddef 实测该算例加载占负载 99.5%)
self.loads[winner] += br.loading + br.computation
```

## 3.4 加权目标

```python
def objective(self, norm_latency_sum, norm_memory_sum, fairness_sum, penalty=0.0):
    # ★ J = ω1·Ψ̃ + ω2·Φ̃ − ω3·Σ_t F(t) + 惩罚 —— 三项【全部无量纲】
    #   用求和(而非平均)是为了与环境的逐时隙奖励【精确】对应
    return (self.w_lat * norm_latency_sum
            + self.w_mem * norm_memory_sum
            - self.w_fair * fairness_sum
            + penalty)
```

> `w_lat = w_mem = w_fair = 1.0` 是默认值。归一化之后这组取值本身就有意义
> （"超一个 deadline" ≈ "占满一台服务器的显存一个时隙" ≈ "完全失衡"），
> 但正式实验仍建议扫一遍权重做敏感性分析。

## 4.1 动作空间（异质，用 `JointAction` 承载）

```python
def _parse_server(self, server_action):
    """(M, K) 保留偏好 φ -> 目标保留层数 (M, K)。"""
    phi = np.clip(np.asarray(server_action, dtype=float), -1.0, 1.0)
    for m in range(self.M):
        for k in range(self.K):
            lo, hi = int(sv.i_check[k]), sv.models[k].N
            # ★ 决策变量①：保留动作 → 目标保留层数
            #   φ=−1 → i_check（卸到 QoS 底线）；φ=+1 → N（全保留）
            #   下界写死为 i_check ⇒ QoS 约束(27e/27g)【由动作映射硬保证】
            targets[m, k] = lo + int(np.ceil((phi[m, k] + 1.0) / 2.0 * (hi - lo)))
    return targets
```

```python
        (self.t % self.epoch) / self.epoch,        # ★ epoch 相位: 服务器状态还有多久会变
    ]
    for m in range(self.M):                        # 每服务器 5 维
        per_server += [rate_m, backhaul_m, f_m, backlog_m, mem_util_m]
    for m in range(self.M):
        for k in range(self.K):                    # 每 (服务器,模型) 3 维
            per_pair += [
                self.states[m, k] / sv.models[k].N,   # ★ 服务器的动作(保留比例)
                self._est_loading_norm(m, k),         # ★★ 按当前保留状态的预估重载/deadline
                1.0 if sv.can_serve(k) else 0.0,      # 可行性掩码
            ]
```

```python
def _est_loading_norm(self, m, k):
    """服务器 m 上模型 k 按当前保留状态重载所需时延 / 该类 deadline。

    这是"把服务器的动作写进用户环境"的核心量: 用户据此判断
    "去这台要不要额外等一次加载"。没有它, 用户分不清
    "模型还在显存里"和"已经被卸了"。
    """
    ld = sv.loading_delay(k, int(self.states[m, k]))
    return float(min(ld / self.sys.deadline_of(k), 1.0))
```

### 4.2.2 服务器观测（8 + 7K 维，K=3 时共 29 维，与 v4 相同）

```python
def _obs_server(self, m):
    backlog = max(0.0, self.busy_until[m] - self.now_ms)   # ★ v4 新增
    glob = [
        self.t / max(self.horizon, 1),             # 时隙进度
        mem_usage / sv.mem_cap_mb,                 # ★ 自身显存利用率
        self.loads[m] / load_sum,                  # ★ 自身相对负载（Jain 相关）
        self.fairness_prev,                        # ★ 上一时隙 Jain
        rate / self._rate_scale,                   # ★ 自身链路速率（信道状态）
        self.sys.channel.backhaul_delay_ms[m] / self._bh_scale,   # 回程时延
        sv.compute_gflop_per_ms / self._f_scale,   # 自身算力
        # ★ v4 新增第 8 维：队列积压（忙至时刻 − 当前时刻），排队建模的必要观测
        #   没有它 agent 看不到"这台机器还要忙多久"，无法做排队感知路由
        min(backlog / max(self.sys.slot_ms, 1e-9), 100.0) / 100.0,
    ]
    for k in range(self.K):
        per_model += [
            self.states[m, k] / n_k,               # ★ 该模型保留比例 s/N
            min(self.idle_ms[m, k] / scale, 1.0),  # ★ 该模型空闲时长
            sv.i_check[k] / n_k,                   # ★ 数学钩子：QoS 下界
            b / self.sys.slot_ms,                  # ★ 数学钩子：ski-rental 临界点 B
            self.arrival_count[k] / elapsed,       # ★ 全局到达率 λ_k
            self.assigned_count[m, k] / elapsed,   # ★ 局部到达率 λ_{m,k}（路由耦合！）
            1.0 if sv.can_serve(k) else 0.0,       # ★ 能否服务（可行性掩码）
        ]
```

> `λ_{m,k}` 是**路由与卸载耦合**的载体，也是**用户策略进入服务器观测的唯一通道**：
> 路由决定它，它又决定该模型的最优卸载时机。

## 4.3 用户路由 + 奖励分账

```python
for task in arrivals:
    k = task.model_id
    eligible = self.sys.eligible_servers(k)        # ★ 只有 can_serve 的服务器可选
    if not eligible:
        slot_penalty += self.sys.w_pen_unserved    # 系统级违规（显存可行性缺口）
        continue
    # ★ 用户路由（连续松弛）：在可服务集合里取偏好最高者 ⇒ x_{j,m*}=1
    #   （2026-08-21 组会：原先是"服务器竞价、最高者得"，已被驳回）
    chosen = max(eligible, key=lambda m: (pref[k, m], -m))
    wait = max(0.0, self.busy_until[chosen] - now)
    br = self.sys.task_latency(task, chosen, int(self.states[chosen, k]),
                               wait_ms=wait, fading_pow=self.fading)
    # ★ 服务后该模型恢复全常驻，必要时驱逐其他模型
    self.states[chosen] = self.sys.servers[chosen].make_room_for(self.states[chosen], k)
```

奖励按**谁控制哪一项**切开 —— 每个 agent 的奖励里只出现它能影响的量：

```python
tmax = task.deadline_ms
# ★ 加载归服务器（层保留唯一控制的时延分量）
srv_norm_lat[chosen] += br.loading / tmax
# ★ 等待/传输/计算归用户（完全由"选了哪台"决定）
user_norm_lat += (br.waiting + br.transmission + br.computation) / tmax
```

```python
# 用户: 每时隙结算
r_user = (- self.sys.w_lat * user_norm_lat        # ★ −(等待+传输+计算)/deadline
          + self.sys.w_fair * fairness            # ★ +Jain（路由的直接后果, 不再除 M）
          - slot_penalty)                         # ★ −罚项（超时/无可服务节点）

# 服务器: 先累进 epoch 累加器, 到 epoch 边界一次性发放
self.server_reward_acc += (- self.sys.w_lat * srv_norm_lat            # ★ −加载/deadline
                          - self.sys.w_mem * mem_usage / self.sys.mem_ref)  # ★ −显存
emit_server = self.is_epoch_start or self.done    # ★ 双时标: 按 epoch 发放
```

> **罚项为什么归用户**：`i_check` 已经用 $D^{\mathrm{QoS}}$ 保证了"钉住的模型重载放得进
> deadline 预算"，所以剩下的超时主要来自排队，而排队是路由的后果。

> **一个不藏的外部性**：预取占 GPU ⇒ 抬高后续任务的**等待**，而等待记在用户账上。
